### Task 5 — RedisSessionRepository

`src/latte_graph_redis/infrastructure/session_repo.py`:
```python
from redis.asyncio import Redis
import functools
import redis.exceptions as redis_exc

from latte_graph_redis.domain.exceptions import RedisUnavailableError, SessionKeyError
from latte_graph_redis.infrastructure.keys import RedisKeyBuilder, SESSION_TTL, MESSAGE_CAP


def _map_redis_errors(fn):
    """Map redis client errors to domain exceptions.

    - `ConnectionError` → `RedisUnavailableError` (fatal — caller must surface).
    - `ResponseError` carrying type mismatch → `SessionKeyError` (non-fatal —
      typically schema drift; caller logs and continues).

    Applied to EVERY public method on RedisSessionRepository so error mapping
    is uniform, not piecemeal per method.
    """
    @functools.wraps(fn)
    async def wrapper(self, *args, **kwargs):
        try:
            return await fn(self, *args, **kwargs)
        except redis_exc.ConnectionError as e:
            raise RedisUnavailableError(str(e)) from e
        except redis_exc.ResponseError as e:
            # WRONGTYPE: e.g. LRANGE on a string key → schema/type drift
            if "WRONGTYPE" in str(e).upper():
                raise SessionKeyError(str(e)) from e
            raise  # other ResponseErrors propagate
    return wrapper


class RedisSessionRepository:
    """Redis-backed session state store for active user sessions.

    All keys follow the ``latte:{user_id}:session:*`` keyspace.
    MESSAGE_CAP = 20 enforced via LTRIM on every append.
    SESSION_TTL = 86400s reset on every write.
    Error handling via ``@_map_redis_errors`` decorator on every public method.

    Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
    Source: PR-204 | BRD.Arch.001 §session-state
    Since: latte-graph-redis 1.0.0
    """

    def __init__(self, client: Redis) -> None:
        """Initialise the repository with an async Redis client.

        Args:
            client: Configured ``redis.asyncio.Redis`` instance (from
                ``RedisClientFactory.create`` or fakeredis in tests).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        self._r = client
        self._k = RedisKeyBuilder

    # --- Messages (List, capped 20) ---
    @_map_redis_errors
    async def append_message(self, user_id: str, message_json: str) -> None:
        """Append a JSON message string to the session message list.

        Enforces MESSAGE_CAP = 20 via LTRIM after every RPUSH.
        Resets SESSION_TTL on the key.

        Args:
            user_id: Session owner.
            message_json: JSON-serialized message string (caller handles serialization).

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        key = self._k.messages(user_id)
        await self._r.rpush(key, message_json)
        await self._r.ltrim(key, -MESSAGE_CAP, -1)
        await self._r.expire(key, SESSION_TTL)

    @_map_redis_errors
    async def get_messages(self, user_id: str) -> list[str]:
        """Return all messages in the session message list (LRANGE 0 -1).

        Args:
            user_id: Session owner.

        Returns:
            List of JSON-serialized message strings; empty list if no messages.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        return await self._r.lrange(self._k.messages(user_id), 0, -1)

    # --- Agent State (Hash) ---
    @_map_redis_errors
    async def set_state(self, user_id: str, fields: dict[str, str]) -> None:
        """Write agent state fields to the session state hash (HSET mapping).

        Resets SESSION_TTL on the key.

        Args:
            user_id: Session owner.
            fields: String-to-string mapping of agent state fields.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        key = self._k.state(user_id)
        await self._r.hset(key, mapping=fields)
        await self._r.expire(key, SESSION_TTL)

    @_map_redis_errors
    async def get_state(self, user_id: str) -> dict[str, str]:
        """Return all agent state fields from the session state hash (HGETALL).

        Args:
            user_id: Session owner.

        Returns:
            String-to-string dict of all hash fields; empty dict if key absent.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        return await self._r.hgetall(self._k.state(user_id))

    # --- String keys (last_query, last_chart, kg_cache) ---
    @_map_redis_errors
    async def set_last_query(self, user_id: str, value: str) -> None:
        """Persist the last natural-language query string (SET with ex=SESSION_TTL).

        Args:
            user_id: Session owner.
            value: Raw query string to store.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        await self._r.set(self._k.last_query(user_id), value, ex=SESSION_TTL)

    @_map_redis_errors
    async def get_last_query(self, user_id: str) -> str | None:
        """Return the last natural-language query string, or None if absent.

        Args:
            user_id: Session owner.

        Returns:
            Stored query string, or ``None`` if key does not exist.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0

        return await self._r.get(self._k.last_query(user_id))

    @_map_redis_errors
    async def add_active_dataset(self, user_id: str, dataset_id: str) -> None:

        await self._r.expire(key, SESSION_TTL)

    @_map_redis_errors
    async def get_pending_kg(self, user_id: str) -> list[str]:
        """Return all pending KG relation strings (LRANGE 0 -1).

        Args:
            user_id: Session owner.

        Returns:
            List of JSON-serialized relation strings; empty list if absent.

        Raises:
            RedisUnavailableError: Redis connection failed.
            SessionKeyError: Key type mismatch (schema drift).

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        return await self._r.lrange(self._k.pending_kg(user_id), 0, -1)

    # --- Session lifecycle ---
    @_map_redis_errors
    async def close_session(self, user_id: str) -> None:
        """Delete all 7 session keys atomically (DEL).

        After this call all GET methods return None/empty for the user.

        Args:
            user_id: Session owner.

        Raises:
            RedisUnavailableError: Redis connection failed.

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        keys = self._k.all_session_keys(user_id)
        if keys:
            await self._r.delete(*keys)

    @_map_redis_errors
    async def reset_ttl(self, user_id: str) -> None:
        """Reset SESSION_TTL on all 7 session keys via a pipelined EXPIRE.

        Uses a non-transactional pipeline to batch all EXPIRE commands
        into a single round-trip.

        Args:
            user_id: Session owner.

        Raises:
            RedisUnavailableError: Redis connection failed.

        Spec: `docs/superpowers/specs/2026-05-19-docstring-traceability-convention-design.md`
        Source: PR-204 | BRD.Arch.001 §session-state
        Since: latte-graph-redis 1.0.0
        """
        # Pipeline EXPIRE calls into one round-trip — same semantics, less latency.
        pipe = self._r.pipeline(transaction=False)
        for key in self._k.all_session_keys(user_id):
            pipe.expire(key, SESSION_TTL)
        await pipe.execute()
```
